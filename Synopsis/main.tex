\section{Abstract}
Pedestrian crowd safety management in dense transit nodes, crosswalks, and narrow corridors presents an urgent urban safety challenge. High mortality rates in crowd crushes stem from physical bottlenecks, footpath encroachments, and uncoordinated pedestrian-vehicular conflicts. Conventional monitoring systems either rely on naive motion subtraction that triggers false alarms during rapid, orderly evacuations, or depend on static shortest-path graph algorithms ($A^*$/Dijkstra) that route escaping crowds directly into saturated choke points.

This project introduces an end-to-end Computer Science framework integrating edge Computer Vision, physical flow mechanics, and heuristic path planning. The perception module segments high-density crowd regions from video frames using multi-source texture fusion (Local Binary Patterns, Fourier analysis, and Gray-Level Co-occurrence Matrix) with a Linear Support Vector Machine. A kinematic flow engine analyzes Optical Flow velocity vectors against a $13^\circ$ directional angular spread threshold to differentiate safe Faster-is-Faster (FIF) egress from fatal Faster-is-Slower (FIS) bottleneck clogging. Finally, a merge-aware evacuation routing engine calculates least-effort escape trajectories while dynamically adjusting pedestrian crossing walk phases.

\section{Problem Definition}
Urban pedestrian infrastructures in metropolitan centers exhibit severe spatial vulnerabilities, where pedestrians account for over $50\%$ of road fatalities due to narrow sidewalks, hawker encroachments, and unsegregated transit hub egress \cite{maitra2024kolkata, morth2023report}. Existing surveillance and routing architectures exhibit three fatal shortcomings:
\begin{enumerate}
    \item \textbf{Motion-Segmentation Fragility:} Motion-differencing and optical flow thresholding generate false positives from passing vehicles and completely fail when crowds remain stationary under extreme compaction \cite{basalamah2020pedestrian, martinez2022computer}.
    \item \textbf{Naive Speed-Based Panic Tripping:} Traditional alerting systems trigger stampede alarms whenever velocity increases, failing to distinguish safe rapid evacuations (Faster-is-Faster) from fatal compression gridlocks (Faster-is-Slower) \cite{liao2017faster, helbing2000simulating}.
    \item \textbf{Suboptimal Shortest-Path Evacuation Routing:} Conventional building evacuation systems route evacuees strictly by Euclidean shortest path, directing crowds directly into right-angle bottlenecks and opposing counterflows \cite{guy2010pledestrians, depta2014self, corbetta2021physics}.
\end{enumerate}

\subsection{Perception and Kinematic State Classification}
Algorithm \ref{alg:state_classification} details the multi-source appearance extraction and flow regime evaluation logic.

\begin{algorithm}[H]
\caption{Multi-Source Crowd Feature Extraction and Regime Evaluation}
\label{alg:state_classification}
\begin{algorithmic}[1]
\Require Video frames $F_t, F_{t-1}$, corridor width $W$, $W_{\text{crit}}=1.0\text{ m}$, $\rho_{\text{crit}}=4.0\text{ ped/m}^2$, $v_{\text{stall}}=0.25\text{ m/s}$, $\theta_{\text{limit}}=13.0^\circ$.
\Ensure HazardState, DensityMap $D_t$, DirectionalSpread $\theta_{\text{spread}}$.
\State Partition current frame $F_t$ into uniform $32\times 32$ pixel cells $C$.
\For{each cell $c \in C$}
    \State $f_{\text{LBP}} \gets \text{ExtractLBP}(c, P=8, R=1)$
    \State $f_{\text{Fourier}} \gets \text{ExtractFourierMoments}(c, \text{cutoff}=0.4)$
    \State $f_{\text{GLCM}} \gets \text{ExtractGLCMProperties}(c, \text{angles}=[0^\circ, 45^\circ, 90^\circ, 135^\circ])$
    \State $\mathbf{x}_c \gets [f_{\text{LBP}} \,\|\, f_{\text{Fourier}} \,\|\, f_{\text{GLCM}}] \in \mathbb{R}^{128}$
    \State $\text{CellScore}[c] \gets \text{LinearSVM\_Predict}(\mathbf{x}_c)$
\EndFor
\State $D_t \gets \text{ApplyGaussianFilter2D}(\text{CellScore}, K_{11\times 11}, \sigma=1.5)$
\State $\rho \gets \text{IntegrateDensity}(D_t) / \text{Area}(\text{ROI})$
\State $\vec{V} \gets \text{ComputeOpticalFlow}(F_{t-1}, F_t, \text{Mask}=D_t)$
\State $v_{\text{avg}} \gets \text{MeanMagnitude}(\vec{V})$, \quad $\theta_{\text{spread}} \gets \sqrt{\frac{1}{N}\sum_{i=1}^N (\theta_i - \bar{\theta})^2}$
\If{$W < W_{\text{crit}} \land \rho \ge \rho_{\text{crit}} \land v_{\text{avg}} \le v_{\text{stall}}$}
    \State \Return ($\text{HazardState.CRITICAL\_CLOG\_FIS}, D_t$)
\ElsIf{$\rho \ge \rho_{\text{crit}} \land v_{\text{avg}} > v_{\text{stall}} \land \theta_{\text{spread}} \le \theta_{\text{limit}}$}
    \State \Return ($\text{HazardState.HIGH\_THROUGHPUT\_FIF}, D_t$)
\ElsIf{$\theta_{\text{spread}} > \theta_{\text{limit}} \land \rho > 2.0$}
    \State \Return ($\text{HazardState.DISORDERED\_STREAM}, D_t$)
\Else
    \State \Return ($\text{HazardState.NORMAL\_FLOW}, D_t$)
\EndIf
\end{algorithmic}
\end{algorithm}

\subsection{Least-Effort Dynamic Evacuation Routing}
Let the corridor network be modeled as a directed graph $G(V, E)$. Grounded in Zipf's Principle of Least Effort \cite{zipf1949human} and the PLEdestrians framework \cite{guy2010pledestrians}, the dynamic edge weight $C(u, v)$ for edge $(u, v)$ with length $d_{uv}$, bottleneck width $W_v$, and merge angle $\Phi_{uv}$ is computed as:
\begin{equation}
C(u, v) = \frac{d_{uv}}{\max(v_{\text{eff}}, 0.1)} + \alpha \cdot D_t[v] + \beta \cdot \left(\frac{1}{\max(W_v, 0.8)}\right) + \gamma \cdot \sin\left(\frac{\Phi_{uv}}{2}\right)
\end{equation}
When target vertex $v$ triggers $\text{HazardState.CRITICAL\_CLOG\_FIS}$, $C(u, v) = \infty$, dynamically forcing priority-queue routing to divert evacuees away from saturated choke points. Furthermore, for urban crossing nodes $k$, pedestrian walk phases are dynamically extended:
\begin{equation}
S_{\text{adjust}}[k].\text{extendWalkPhaseBy} = \min(15.0, D_t[k] \times 2.5)\text{ seconds}
\end{equation}
